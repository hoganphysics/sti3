#include <filesystem>
#include <chrono>
#include <random>
#include <string>

struct TempDir {
  std::filesystem::path path;

  TempDir(std::string prefix = "myproj-test-") {
    auto base = std::filesystem::temp_directory_path();

    // simple unique suffix
    auto now = std::chrono::high_resolution_clock::now().time_since_epoch().count();
    std::mt19937_64 rng(static_cast<uint64_t>(now));
    auto r = rng();

    path = base / (prefix + std::to_string(now) + "-" + std::to_string(r));
    std::filesystem::create_directories(path);
  }

  ~TempDir() {
    std::error_code ec;
    std::filesystem::remove_all(path, ec); // ignore errors in destructor
  }
};
